Nível 2 JOIN + GROUP BY
Consulta em estudo 
SELECT c.nome, COUNT(p.id) AS pedidos
FROM clientes c
JOIN pedidos p ON p.cliente_id = c.id
GROUP BY c.nome
ORDER BY pedidos DESC;
